cmake_minimum_required(VERSION 3.16)
project(iris_vio_quickstart LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GZ_TRANSPORT REQUIRED IMPORTED_TARGET gz-transport13)
pkg_check_modules(GZ_MSGS REQUIRED IMPORTED_TARGET gz-msgs10)
find_package(OpenCV REQUIRED)
find_package(Threads REQUIRED)
add_executable(vio_tracker vio_tracker.cpp)
target_link_libraries(vio_tracker PRIVATE PkgConfig::GZ_TRANSPORT PkgConfig::GZ_MSGS ${OpenCV_LIBS} Threads::Threads)
target_include_directories(vio_tracker PRIVATE ${OpenCV_INCLUDE_DIRS})
